CREATE DATABASE audio_text_db;
USE audio_text_db;
++++++++++++++++++++++++++++++++++++++++
CREATE TABLE conversions (

    id INT AUTO_INCREMENT PRIMARY KEY,

    conversion_type VARCHAR(50) NOT NULL,

    input_text TEXT,

    output_text TEXT,

    language VARCHAR(50),

    audio_file VARCHAR(255),

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP

);
